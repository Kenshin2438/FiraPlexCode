// Async data pipeline with arrow ligatures and fancy operators
import { fetchUser, type User } from "./api";

async function main(ids: number[]): Promise<User[]> {
  /* Italic block: comments are styled by IBM Plex Mono Italic.
     Look at how f, a, r, k, x flow differently from upright. */
  const users = await Promise.all(ids.map(async id => {
    const u = await fetchUser(id);
    return u ?? { id, name: "unknown" };
  }));
  return users.filter(u => u.id !== 0 && u.id <= 9999);
}
